\newcommand{\SPcard}[7]{%
  \begin{scope}[xshift=#1,yshift=#2]

    % Background
    \fill[white]
      (0,0) rectangle (\SPwidth,-\SPheight);

    % Colored definition header
    \fill[#3]
      (0,0) rectangle (\SPwidth,-\SPhead);

    % Border
    \draw[SPline,line width=.45pt]
      (0,0) rectangle (\SPwidth,-\SPheight);

    % Divider
    \draw[SPline,line width=.35pt]
      (0,-\SPhead) -- (\SPwidth,-\SPhead);

    % Condition title
    \node[
      anchor=north west,
      inner sep=0pt,
      text width=\SPtext,
      align=left,
      font=\sffamily\bfseries\fontsize{8.8}{10.5}\selectfont
    ] at (8pt,-7pt) {#4};

    % Definition
    \node[
      anchor=north west,
      inner sep=0pt,
      text width=\SPtext,
      align=left,
      font=\rmfamily\fontsize{8.3}{10}\selectfont
    ] at (8pt,-23pt) {#5};

    % Example title
    \node[
      anchor=north west,
      inner sep=0pt,
      text width=\SPtext,
      align=left,
      font=\rmfamily\bfseries\fontsize{8.5}{10}\selectfont
    ] at (8pt,-57pt) {#6};

    % Example text
    \node[
      anchor=north west,
      inner sep=0pt,
      text width=\SPtext,
      align=left,
      font=\rmfamily\fontsize{8.1}{9.8}\selectfont
    ] at (8pt,-75pt) {#7};

  \end{scope}%
}
